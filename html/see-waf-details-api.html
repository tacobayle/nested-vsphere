jsonFile=json/nic-vsphere-nsx-avi_apply.json
source bash/variables.sh
# display variables
echo ${avi_version}
echo ${avi_username}
echo ${ip_avi}
curl -s -k -X POST -H "Content-Type: application/json" -d "{\"username\": \"${avi_username}\", \"password\": \"${GENERIC_PASSWORD}\"}" -c avi_cookie.txt https://${ip_avi}/login
# show default WAF policy
curl -s -k -X GET -H "Content-Type: application/json" -H "X-Avi-Version: ${avi_version}" -b avi_cookie.txt https://${ip_avi}/api/wafpolicy | jq '.results[] | select(.name == "System-WAF-Policy")'
wafcrs_uuid=$(curl -s -k -X GET -H "Content-Type: application/json" -H "X-Avi-Version: ${avi_version}" -b avi_cookie.txt https://${ip_avi}/api/wafpolicy | jq -c -r '.results[] | select(.name == "System-WAF-Policy").waf_crs_ref' | cut -d"/" -f6)
# show WAF CRS uuid
echo ${wafcrs_uuid}
# show specific CRS rule
curl -s -k -X GET -H "Content-Type: application/json" -H "X-Avi-Version: ${avi_version}" -b avi_cookie.txt https://${ip_avi}/api/wafcrs/${wafcrs_uuid} | jq '.groups[] | select(.name == "CRS_942_Application_Attack_SQLi").rules[] | select(.name =="Check for SQL Injection (1/35)")'
